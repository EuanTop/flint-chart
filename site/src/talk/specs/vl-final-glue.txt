import embed from 'vega-embed';
import { changeset } from 'vega';
import { showRows, gamePicker } from './ui.js';

const UNIT = 'concat_1'; // the heatmap's internal view name

export async function mount(el, spec, rows) {
  const { view } = await embed(el, spec, { actions: false });

  // chart -> table: the 'pick' signal holds { game: [...] } or {}
  view.addSignalListener('pick', (_, sel) => {
    const games = sel.game ?? [];
    showRows(rows.filter((r) => games.includes(r.game)));
  });

  // picker -> chart: write a point tuple into the selection's internal store
  gamePicker(rows, async (game) => {
    const fields = [{ type: 'E', field: 'game' }];
    const tuples = game ? [{ unit: UNIT, fields, values: [game] }] : [];
    const change = changeset().remove(() => true).insert(tuples);
    await view.change('pick_store', change).runAsync();
  });
  return view;
}
